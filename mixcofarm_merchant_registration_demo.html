const form = document.getElementById('registrationForm'); const result = document.getElementById('result'); const preview = document.getElementById('statusPreview'); const submitButton = form.querySelector('button[type="submit"]'); form.addEventListener('submit', async (e) => { e.preventDefault(); if (!form.reportValidity()) return; const data = Object.fromEntries(new FormData(form).entries()); data.consent = document.getElementById('consent').checked; data.city = data.location || ''; delete data.location; result.classList.remove('hidden'); result.textContent = 'Sending demo registration to the server...'; submitButton.disabled = true; try { const response = await fetch( 'https://babujiontop-github-io.onrender.com/api/merchant/registrations', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) } ); const reply = await response.json(); if (!response.ok || reply.saved !== true) { throw new Error(reply.error || 'Registration could not be saved.'); } result.replaceChildren(); const heading = document.createElement('h2'); heading.textContent = 'Registration received'; const reference = document.createElement('p'); reference.textContent = 'Reference: ' + reply.reference; const status = document.createElement('span'); status.className = 'statuspill'; status.textContent = 'Pending demo review'; const note = document.createElement('p'); note.className = 'muted small'; note.textContent = 'Your demo application was saved by the server. It has not been verified or approved.'; result.append(heading, reference, status, note); preview.replaceChildren(); [ ['Business: ', data.store], ['Category: ', data.category], ['Country: ', data.country] ].forEach(([label, value]) => { const p = document.createElement('p'); p.textContent = label + (value || 'Not provided'); preview.appendChild(p); }); const previewStatus = document.createElement('span'); previewStatus.className = 'statuspill'; previewStatus.textContent = 'Pending demo review'; preview.appendChild(previewStatus); } catch (error) { result.replaceChildren(); const heading = document.createElement('h2'); heading.textContent = 'Registration not confirmed'; const message = document.createElement('p'); message.textContent = error.message || 'Could not contact the server. Please try again.'; result.append(heading, message); } finally { submitButton.disabled = false; result.scrollIntoView({ behavior: 'smooth', block: 'nearest' }); } }); document.getElementById('resetBtn').addEventListener('click', () => { result.classList.add('hidden'); preview.textContent = 'No demo registration submitted yet.'; });
